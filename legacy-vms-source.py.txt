# Helper to convert HH:MM string to float hours (e.g., "08:30" -> 8.5)
def time_to_float(time_str):
    try:
        if isinstance(time_str, (int, float)):
            return float(time_str)
        # Handle cases like "50:" which might occur if split is on a non-standard time format
        if not time_str or ':' not in time_str:
            return 0.0
        h, m = map(int, time_str.split(':'))
        return h + m / 60.0
    except:
        return 0.0

# Helper to convert float hours to HH:MM string (e.g., 8.75 -> "08:45")
def float_to_time(float_hours):
    total_minutes = int(round(float_hours * 60))
    hours = total_minutes // 60
    minutes = total_minutes % 60
    return f"{hours:02d}:{minutes:02d}"

class SortableTableWidget(QtWidgets.QTableWidget):
    """QTableWidget that allows sorting by clicking headers and implements search filtering."""
    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self.horizontalHeader().sectionClicked.connect(self.sort_by_column)
        # Enable initial sorting indicator
        self.setSortingEnabled(True)
        # ENFORCE FIXED EDITING MODE
        self.setEditTriggers(QtWidgets.QAbstractItemView.EditTrigger.NoEditTriggers)

    def sort_by_column(self, index):
        # Disable sorting to prevent re-sort on data load (handled by setRowCount)
        self.setSortingEnabled(False)
        
        # Get current order and section
        current_index = self.horizontalHeader().sortIndicatorSection()
        current_order = self.horizontalHeader().sortIndicatorOrder()
        
        # Determine new order
        if current_index != index:
            new_order = QtCore.Qt.SortOrder.AscendingOrder
        else:
            new_order = QtCore.Qt.SortOrder.DescendingOrder if current_order == QtCore.Qt.SortOrder.AscendingOrder else QtCore.Qt.SortOrder.DescendingOrder
        
        # Perform sort and set indicator
        self.sortItems(index, new_order)
        self.horizontalHeader().setSortIndicator(index, new_order)
        
        self.setSortingEnabled(True)

    def setRowCount(self, count):
        self.setSortingEnabled(False)
        super().setRowCount(count)
        self.setSortingEnabled(True)
        
    def filter_data(self, search_text):
        search_text = search_text.lower()
        self.setSortingEnabled(False)
        for row in range(self.rowCount()):
            match = False
            for col in range(self.columnCount()):
                item = self.item(row, col)
                # Check for a match in the cell's text
                if item and search_text in item.text().lower():
                    match = True
                    break
            self.setRowHidden(row, not match)
        self.setSortingEnabled(True)

class DatabaseManager:

        os.makedirs(os.path.dirname(self.db_path), exist_ok=True)
        self._create_tables()

    def _get_conn(self):
        return sqlite3.connect(self.db_path)

    def _create_tables(self):
        conn = self._get_conn()
        cur = conn.cursor()

        # Users table (UNCHANGED)
        cur.execute("""
            CREATE TABLE IF NOT EXISTS users (

                name TEXT NOT NULL,

                role TEXT NOT NULL,
                password TEXT NOT NULL

        try:
            cur.execute("SELECT sql FROM sqlite_master WHERE type='table' AND name='operator_entries';")

            pass

    def update_operator_entry(self, entry_id, estimated_time, time_taken, status, remarks):
        conn = self._get_conn()
        cur = conn.cursor()
        
        # Recalculate Production/Extra hours based on new time_taken value
        taken_hours_float = time_to_float(time_taken)
        max_prod_hours = 8.5
        
        if taken_hours_float <= max_prod_hours:
            prod_hours_str = float_to_time(taken_hours_float) if taken_hours_float > 0 else "00:00"
            extra_hours_str = "00:00"
        else:
            prod_hours_str = float_to_time(max_prod_hours)
            extra_hours_float = taken_hours_float - max_prod_hours
            extra_hours_str = float_to_time(extra_hours_float)
            
        cur.execute("""
            UPDATE operator_entries
            SET estimated_time=?, time_taken=?, status=?, remarks=?, production_hours=?, extra_hours=?
            WHERE id=?;
        """, (estimated_time, time_taken, status, remarks, prod_hours_str, extra_hours_str, entry_id)); conn.commit(); conn.close()

                "operator_id": emp_id,
                "project_name": project_name,
                "model_no": model_no,
                "entry_count": len(rows),
                "latest_estimated_time": float_to_time(latest_est_time),
                "total_estimated_time_float": total_estimated_time_float, # Fixed: Should be the single target time
                "total_time_taken_float": total_time_taken_float,
            }
            
            # 4. Calculate performance status
            if total_estimated_time_float > 0 and total_time_taken_float > 0:
                diff = total_estimated_time_float - total_time_taken_float
                
                # Performance status based on difference
                if diff >= total_estimated_time_float * 0.2: 
                    performance_data["status"] = "Excellent (>= 20% faster)"
                elif diff > 0: 
                    performance_data["status"] = "Good (Faster)"
                elif abs(diff) <= total_estimated_time_float * 0.05:
                    performance_data["status"] = "On Track (Matches Est.)"
                elif diff > total_estimated_time_float * -0.2:
                    performance_data["status"] = "Fair (Up to 20% slower)"
                else:
                    performance_data["status"] = "Needs Improvement (> 20% slower)"
            elif total_time_taken_float > 0 and total_estimated_time_float == 0:
                performance_data["status"] = "Time Logged (Est. Missing)"
            else:
                performance_data["status"] = "No Data"

            return performance_data

        finally:
            conn.close()

            FROM operator_entries 
            WHERE project_name = ? AND model_no != '' 
            AND status != 'DELETED' AND deleted_time_log = '00:00' 
            ORDER BY model_no;

            QtWidgets.QMessageBox.warning(self, "Missing Selection", "Please select an Operator, Project, and Model.")

        try:
            data = self.db.calculate_operator_performance_data(emp_id, project_name, model_no)

            QtWidgets.QMessageBox.critical(self, "Database Error", f"Failed to load performance data: {e}")
            return

    # Admin's Project Excel Upload Logic (UNCHANGED)
    def upload_project_excel(self):
        path, _ = QtWidgets.QFileDialog.getOpenFileName(
            self,
            "Select Project Excel File",
            "",
            "Excel Files (*.xlsx *.xls)"
        )
        if not path:
            return

        try:
            # Check if file exists and is readable (precautionary)
            if not os.path.exists(path):
                raise FileNotFoundError(f"File not found: {path}")

            # Added engine='openpyxl' for robust modern Excel file reading
            # Ensure openpyxl is installed: pip install pandas openpyxl
            df = pd.read_excel(path, sheet_name=0, engine='openpyxl')
            
            # Clean column names by stripping whitespace
            df.columns = df.columns.str.strip() 

            required_cols = ["Project Name", "Estimated Time (hours)", "Module Number", "Delivery Date"]
            
            if not all(col in df.columns for col in required_cols):
                raise ValueError("Missing essential columns: Project Name, Estimated Time (hours), Module Number, Delivery Date. Please check column headers.")

            if "Start Date" not in df.columns:
                df["Start Date"] = datetime.now().strftime("%Y-%m-%d")
            
            # Use fillna('') to handle missing values robustly before grouping/string operations
            df = df.fillna('')
            
            project_groups = df.groupby("Project Name")
            
            total_projects_processed = 0
            
            for name, group in project_groups:
                project_name = str(name).strip()
                if not project_name: continue
                
                total_projects_processed += 1

                # Find overall estimates (if present) for total project summary fields
                total_est_row = group[group["Module Number"].astype(str).str.contains("Overall Estimate", na=False)]
                
                if not total_est_row.empty:
                    # Use .iloc[0] to safely access the first value from the filtered row
                    total_estimated_hours = float(total_est_row["Estimated Time (hours)"].iloc[0]) if total_est_row["Estimated Time (hours)"].iloc[0] != '' else 0.0
                    delivery_date = total_est_row["Delivery Date"].iloc[0]
                    start_date = total_est_row["Start Date"].iloc[0]
                else:
                    # Calculate totals from module rows if overall estimate is missing
                    module_rows = group[~group["Module Number"].astype(str).str.contains("Total|Overall Estimate", na=False)]
                    
                    # Convert to numeric first, handling non-numeric data as NaN, then fill NaN with 0 for sum
                    numeric_module_hours = pd.to_numeric(module_rows["Estimated Time (hours)"], errors='coerce').fillna(0)
                    total_estimated_hours = numeric_module_hours.sum()
                    
                    delivery_date = group["Delivery Date"].max()
                    start_date = group["Start Date"].min()

                # Format dates robustly
                try:
                    delivery_date_str = pd.to_datetime(delivery_date).strftime("%Y-%m-%d")
                except:
                    delivery_date_str = str(delivery_date).split(' ')[0] if delivery_date else ''
                    
                try:
                    start_date_str = pd.to_datetime(start_date).strftime("%Y-%m-%d")
                except:
                    start_date_str = str(start_date).split(' ')[0] if start_date else ''


                for _, row in group.iterrows():
                    module_number = str(row["Module Number"]).strip()
                    
                    if "Overall Estimate" in module_number or "Total" in module_number or not module_number:
                        continue
                    
                    model_no = module_number.split('(')[0].strip() 
                    
                    module_hours_raw = row["Estimated Time (hours)"]
                    
                    # Ensure module hours is a float for time_to_float
                    if module_hours_raw == '':
                        module_hours_float = 0.0
                    else:
                        try:
                            module_hours_float = float(module_hours_raw)
                        except (ValueError, TypeError):
                            module_hours_float = 0.0 # Default to 0 if conversion fails
                        
                    module_estimated_time_str = float_to_time(module_hours_float) 

                    # Insert or Replace Module Data
                    self.db.insert_project_data_from_excel(
                        project_name, 
                        total_estimated_hours, 
                        delivery_date_str, 
                        start_date_str, 
                        model_no, 
                        module_estimated_time_str
                    )
            
            # Refresh projects tab after successful import
            self.refresh()
            QtWidgets.QMessageBox.information(self, "Success", f"Project data from Excel imported and refreshed successfully. ({total_projects_processed} unique projects processed)")

        except Exception as e:
            QtWidgets.QMessageBox.critical(self, "Error", f"Failed to import Excel. Check file format or column content:\n{e}")

    def submit_entry(self):

        model = self.cb_model.currentText()
        
        est_time_str = f"{self.cb_est_h.currentText()}:{self.cb_est_m.currentText()}"

    # -------- LOAD TODAY HISTORY (MODIFIED: Project dropdown population and Status display) --------
    def load_history(self):
        conn = self.db._get_conn()
        cur = conn.cursor()
        today = self.date.date().toString("yyyy-MM-dd")
        
        # --- Update Project Dropdown: Only show non-dispatched projects + Default tasks ---
        self.cb_project.blockSignals(True)
        self.cb_project.clear()
        self.cb_project.addItem("Select Project")
        
        # Add non-dispatched projects
        active_projects = self.db.get_non_dispatched_projects_for_dropdown()
        
        # Add non-project activities
        default_projects = ["Office meeting", "Celebration", "System problem", 
                            "Training", "Power cut", "Permission", "Late", "Project discussion", "Half day Leave"]
        
        combined_projects = sorted(active_projects + default_projects)
        self.cb_project.addItems(combined_projects)
        self.cb_project.blockSignals(False)


        # Query all necessary fields including Prod Hrs and Extra Hrs for calculation (Excluding deleted)
        cur.execute("""
            SELECT date, project_name, model_no, estimated_time,
                     time_taken, production_hours, extra_hours, status, remarks
            FROM operator_entries
            WHERE emp_id = ? AND date = ? AND status != 'DELETED' AND deleted_time_log = '00:00'
            ORDER BY id DESC;
        """, (self.user["emp_id"], today))

        rows = cur.fetchall()
        conn.close()
        
        # --- Calculate remaining time for header display ---
        # Fetching production_hours and extra_hours strings directly from DB logs
        total_prod_consumed_float = sum(time_to_float(row[5]) for row in rows)
        total_extra_float = sum(time_to_float(row[6]) for row in rows)
        
        max_capacity_float = 8.5 # 8 hours 30 minutes
        remaining_float = max(0.0, max_capacity_float - total_prod_consumed_float)
        remaining_time_str = float_to_time(remaining_float)
        
        # FIX: Ensure QTimeEdit is set correctly from HH:MM string components
        h_rem, m_rem = map(int, remaining_time_str.split(':'))
        self.prod.setTime(QtCore.QTime(h_rem, m_rem))
        
        total_extra_str = float_to_time(total_extra_float)
        h_extra, m_extra = map(int, total_extra_str.split(':'))
        self.extra.setTime(QtCore.QTime(h_extra, m_extra))


        # --- Populate Table ---
        self.history_table.setRowCount(len(rows))
        
        for r, row in enumerate(rows):
            # Display Row (7 columns): [0:Date, 1:Project, 2:Model, 3:Estimated, 4:Taken, 5:Status, 6:Remarks]
            display_row = list(row[0:5]) + [row[7]] + [row[8]]
            
            # Convert status 'Comp' to 'Complete' for display
            if display_row[5] == 'Comp': display_row[5] = 'Complete'

            for c, v in enumerate(display_row):
                self.history_table.setItem(r, c, QtWidgets.QTableWidgetItem(str(v)))
                
        self.history_table.horizontalHeader().resizeSections(QtWidgets.QHeaderView.ResizeMode.ResizeToContents) # Force resize after data load
        self.history_table.horizontalHeader().setSectionResizeMode(6, QtWidgets.QHeaderView.ResizeMode.Stretch) # Remarks column stretch

    def refresh_project_dropdowns(self):
        # Update project data and reload history (which updates the dropdown)
        self.project_data = self._load_project_data()
        self.load_history()

    # NEW LOGIC: Auto populate fields when a project is selected (UNCHANGED)
    def _auto_populate_project_data(self):
        project_name = self.p_name.currentText()
        
        # Reset fields
        self.start_date.setDate(QtCore.QDate.currentDate())
        self.delivery.setDate(QtCore.QDate.currentDate())
        self.est_hours.setValue(0.0)
        self.hours_taken.setValue(0.0)
        self.hours_diff.setValue(0.0)
        
        if project_name == "Select Project" or not project_name:
            return

        if project_name in self.project_details:
            details = self.project_details[project_name]

            # 1. Estimated Hours (Total)
            est_hours_float = details['est_hours']
            self.est_hours.setValue(est_hours_float)

            # 2. Start Date
            try:
                start_date_q = QtCore.QDate.fromString(details['start_date'], "yyyy-MM-dd")
                if start_date_q.isValid():
                    self.start_date.setDate(start_date_q)
            except:
                pass
                
            # 3. Delivery Date (Used for Delivery Schedule field)
            try:
                delivery_date_q = QtCore.QDate.fromString(details['delivery_date'], "yyyy-MM-dd")
                if delivery_date_q.isValid():
                    self.delivery.setDate(delivery_date_q)
            except:
                pass


            # 4. Total Time Taken (Aggregated from operator_entries by ALL operators)
            # Fetch the sum of time_taken from all operators for this project
            total_taken_float = self.db.calculate_total_time_taken_for_project(project_name)
            self.hours_taken.setValue(total_taken_float)

            # 5. Calculate Difference
            self._update_hours_diff()
            
        # Note: If project not in details, fields remain reset (0/current date)

    # LOGOUT FUNCTION (UNCHANGED)

        if self.logout_callback:
            self.logout_callback()

    def _update_hours_diff(self):
        # Calculation: |Estimated Hours - Hours Taken| to show magnitude of difference
        diff = abs(self.est_hours.value() - self.hours_taken.value())
        self.hours_diff.setValue(diff)

    def save_entry(self):
        proj = self.p_name.currentText().strip()
        
        if proj == "Select Project":
            QtWidgets.QMessageBox.warning(self, "Missing", "Please select a Project.")
            return
        
        # NEW: Check for duplicate dispatched entry
        if self.db.check_project_dispatched(proj):
            QtWidgets.QMessageBox.critical(self, "Error", f"Project '{proj}' is already marked as Dispatched. Cannot create a new entry.")
            return
            
        start = self.start_date.date().toString("yyyy-MM-dd")
        est = float(self.est_hours.value())
        delivery = self.delivery.date().toString("yyyy-MM-dd")
        
        # LOGIC FIX: Check dispatched status and normalize to 'YES'
        disp = self.dispatched.text().strip()
        disp_save = 'YES' if disp and disp.lower() in ['yes', 'y'] else disp
            
        taken = float(self.hours_taken.value())
        diff = float(self.hours_diff.value()) 
        inv = self.invoice_date.date().toString("yyyy-MM-dd")
        pay = self.payment_date.date().toString("yyyy-MM-dd")

        self.db.insert_accounts_entry(
            proj, start, est, delivery, disp_save, taken, diff, inv, pay
        )
        QtWidgets.QMessageBox.information(self, "Saved", "Accounts entry saved.")
        self.refresh()

        # Resetting fields manually after successful save
        self.p_name.setCurrentIndex(0) 
        self.dispatched.clear()
        self.est_hours.setValue(0.0)
        self.hours_taken.setValue(0.0)
        self.hours_diff.setValue(0.0)

    def edit_accounts_entry(self, record):
        """Handles the editing of an accounts entry."""
        dlg = AccountsEditDialog(self, record, self.project_details)
        if dlg.exec() == QtWidgets.QDialog.DialogCode.Accepted:
            (entry_id, proj, start, est, delivery, disp, taken, diff, inv, pay) = dlg.get_values()
            
            # Check if the updated entry makes a dispatched project un-dispatched, 
            if disp == 'YES' and self.db.is_project_dispatched(proj, exclude_id=entry_id):
                 QtWidgets.QMessageBox.critical(self, "Error", f"Project '{proj}' is already dispatched by another entry. Please verify dispatch history.")
                 return
                
            try:
                self.db.update_accounts_entry(entry_id, proj, start, est, delivery, disp, taken, diff, inv, pay)
                QtWidgets.QMessageBox.information(self, "Success", "Accounts entry updated successfully.")
                self.refresh()
                # Global refresh call to update dropdowns in other dashboards (Operator/PL)
                if isinstance(self.parent(), MainWindow):
                    self.parent().refresh_project_dropdowns()
            except Exception as e:
                QtWidgets.QMessageBox.critical(self, "Error", f"Failed to update entry: {e}")

    def delete_accounts_entry(self, entry_id):
        """Handles the deletion of an accounts entry."""
        reply = QtWidgets.QMessageBox.question(
            self,
            "Confirm Delete",
            "Are you sure you want to permanently delete this accounts entry?",
            QtWidgets.QMessageBox.StandardButton.Yes | QtWidgets.QMessageBox.StandardButton.No
        )
        if reply == QtWidgets.QMessageBox.StandardButton.Yes:
            self.db.delete_accounts_entry(entry_id)
            QtWidgets.QMessageBox.information(self, "Deleted", "Accounts entry permanently deleted.")
            self.refresh()
            # Global refresh call to update dropdowns in other dashboards (Operator/PL)
            if isinstance(self.parent(), MainWindow):
                self.parent().refresh_project_dropdowns()

    # Added for global refresh call
    def refresh_project_dropdowns(self):
        # Update internal project details and refresh dropdown lists
        self.project_details = self._load_project_details()
        self.refresh()

            return AdminDashboard(self.db, logout_callback=self.handle_logout)
        elif role == "Project Leader":
            return ProjectLeaderDashboard(self.db, self.user, logout_callback=self.handle_logout)
        elif role == "Accounts":
            return AccountsDashboard(self.db, logout_callback=self.handle_logout)
        else:
            return OperatorDashboard(self.db, self.user, logout_callback=self.handle_logout)
            
    def handle_logout(self):
        self.login_win = LoginWindow(self.db)
        self.login_win.show()
        self.close()

    def refresh_project_dropdowns(self):
        """
        Global method to trigger the refresh of project lists 
        in active dashboards (Operator/PL/Accounts) after a change in Accounts.
        """
        # Ensure the current widget has the necessary refresh method
        if hasattr(self.current_widget, 'refresh_project_dropdowns'):
            self.current_widget.refresh_project_dropdowns()

# ===================== SPLASH SCREEN (UNCHANGED) ======================
def show_splash(app):
    pix = QtGui.QPixmap(420, 200)
    pix.fill(QtGui.QColor("#10793F"))

    painter = QtGui.QPainter(pix)
    painter.setPen(QtGui.QColor("white"))
    font = QtGui.QFont("Segoe UI", 22, QtGui.QFont.Weight.Bold)
    painter.setFont(font)
    painter.drawText(pix.rect(), QtCore.Qt.AlignmentFlag.AlignCenter, "VMS TOOL")
    painter.end()

    splash = QtWidgets.QSplashScreen(pix)
    splash.show()
    app.processEvents()
    QtCore.QThread.msleep(1500)
    splash.close()
